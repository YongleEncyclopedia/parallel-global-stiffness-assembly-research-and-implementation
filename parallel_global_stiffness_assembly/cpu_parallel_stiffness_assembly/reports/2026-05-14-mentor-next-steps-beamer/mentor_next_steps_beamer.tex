\begin{frame}{两阶段组装术语：把已有 C++ 路线重新命名清楚}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{符号组装}
\begin{itemize}
  \item 处理拓扑、DOF、CSR 稀疏结构。
  \item 预计算每个单元局部矩阵项写入全局 CSR 的 scatter 位置。
  \item 不计算 \texttt{Ke}。
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{数值组装 / 物理组装}
\begin{itemize}
  \item 计算 \texttt{physics\_tet4} 单元刚度矩阵 \texttt{Ke}。
  \item 复用 CSR pattern 和 scatter plan。
  \item 只填充全局矩阵 values。
\end{itemize}
\end{column}
\end{columns}

\vspace{0.8em}
\begin{alertblock}{对照对象：无符号直接组装}
不复用 CSR pattern 或 scatter plan；每次从单元 DOF 直接生成 \texttt{(row,col,value)} 贡献，再排序归并为全局矩阵。
\end{alertblock}
\end{frame}

\begin{frame}{符号/无符号评估设置}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.47\textwidth}
\textbf{Case}
\begin{itemize}
  \item \texttt{3d-WindTurbineHub}
  \item nodes = \metric{228384}
  \item elements = \metric{1113684}
  \item dofs = \metric{685152}
  \item element = \texttt{Tet4}
  \item kernel = \texttt{physics\_tet4}
\end{itemize}
\end{column}
\begin{column}{0.49\textwidth}
\textbf{Platform}
\begin{itemize}
  \item Apple M4 Max
  \item macOS arm64
  \item Clang 21.0.0
  \item OpenMP 202011
  \item physical cores = 14, logical cores = 14
\end{itemize}
\end{column}
\end{columns}
\begin{block}{解释口径}
\texttt{assemblies=1} 表示单次组装总成本；\texttt{assemblies>1} 表示同一稀疏结构下多次数值/物理组装时，符号组装成本被摊销后的总成本。
\end{block}
\end{frame}

\begin{frame}{回答 (1)：单次组装 vs 多次组装}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{单次组装}
\begin{itemize}
  \item 符号复用路径总耗时：\metric{3818.786 ms}
  \item 无符号直接路径：\metric{6276.199 ms}
  \item 单次也能快 \metric{1.644x}
  \item 原因：直接路径的排序归并成本非常高。
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{多次组装}
\begin{itemize}
  \item 3 次：\metric{3.565x}
  \item 10 次：\metric{6.320x}
  \item 30 次：\metric{8.086x}
  \item 多次 Newton iteration / time stepping 场景更应该复用符号结构。
\end{itemize}
\end{column}
\end{columns}
\begin{alertblock}{汇报结论}
固定网格、固定 DOF 布局、固定稀疏结构时，符号组装的价值主要来自“一次构建、多次复用”。
\end{alertblock}
\end{frame}

\begin{frame}{回答 (2)：有符号组装 vs 无符号直接组装}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
\textbf{有符号组装}
\begin{itemize}
  \item CSR pattern 固定。
  \item 单元 DOF 缓存。
  \item \texttt{Ke(i,j)} 到 CSR value index 的 scatter 缓存。
  \item 数值阶段不再做全局排序归并。
\end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
\textbf{无符号直接组装}
\begin{itemize}
  \item 每次生成所有 \texttt{(row,col,value)}。
  \item 每次排序并 reduce。
  \item WindHub case 单次排序归并约 \metric{5.5 s}。
  \item transient memory 约 \metric{2.39 GiB}。
\end{itemize}
\end{column}
\end{columns}
\begin{block}{要点}
当前实验不是简单比较“两个函数名”，而是在比较“是否复用全局稀疏结构与 scatter 写入位置”。
\end{block}
\end{frame}

\begin{frame}{核函数边界：不要混淆 mesh topology 和 kernel mode}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{Mesh / element topology}
\begin{itemize}
  \item \texttt{tet4} / Abaqus \texttt{C3D4}
  \item \texttt{hex8} / Abaqus \texttt{C3D8}
  \item 决定单元节点数与 DOF mapping。
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Kernel mode}
\begin{itemize}
  \item \texttt{simplified}: synthetic local matrix。
  \item \texttt{physics\_tet4}: 基于体积、形函数矩阵和 $B^TDB$ 的 Tet4 物理刚度。
  \item \texttt{physics\_tet4} 对非 Tet4 会回退到 simplified。
\end{itemize}
\end{column}
\end{columns}
\begin{alertblock}{汇报口径}
\texttt{simplified} 用来隔离 assembly overhead；\texttt{physics\_tet4} 用来支撑真实 Tet4/C3D4 工程验证。两类结论要分开讲。
\end{alertblock}
\end{frame}

\begin{frame}{并行算法的硬件适配：本项目如何设定比较}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
\textbf{线程与 OpenMP 环境}
\begin{itemize}
  \item default：清空 \texttt{OMP\_DYNAMIC} / \texttt{OMP\_PROC\_BIND} / \texttt{OMP\_PLACES}。
  \item bound：\texttt{OMP\_DYNAMIC=FALSE}, \texttt{OMP\_PROC\_BIND=close}, \texttt{OMP\_PLACES=cores}。
  \item 比较 timing、speedup、efficiency、extra memory。
\end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
\textbf{核心 profile}
\begin{itemize}
  \item Intel: Linux \texttt{taskset} 做 P-core / E-core affinity restriction。
  \item Apple: macOS QoS-biased profile，不是硬绑核。
  \item 两个平台都要避免把系统调度差异写成算法本质差异。
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{当前能回答与还需要继续推进的问题}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
\textbf{已经能回答}
\begin{itemize}
  \item 单次与多次组装：符号复用收益清楚。
  \item 有符号与无符号：直接排序归并是主要成本。
  \item 不同硬件 profile：已有 Intel P/E 与 Apple QoS sensitivity 图。
  \item 物理核/逻辑核：当前没有真实 SMT 逻辑核收益。
\end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
\textbf{后续应继续做}
\begin{itemize}
  \item 符号组装并行化：当前 CSR/scatter 构建仍是核心预处理成本。
  \item 符号与数值并行解耦：先定义 artifact 生命周期，再比较 pipeline。
  \item 更多真实平台：补有 SMT 的 CPU，才能严谨回答逻辑核收益。
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{结论}
\begin{enumerate}
  \item \keyword{二阶段组装已经在当前 C++ 主线中落地}：CSR sparsity + scatter plan 是符号阶段，\texttt{Ke} 计算与 value 填充是数值阶段。
  \item \keyword{单次组装也受益，多次组装收益更强}：WindHub + \texttt{physics\_tet4} 下，符号复用相对无符号直接组装从 \metric{1.644x} 提升到 \metric{8.086x}。
  \item \keyword{无符号直接组装的瓶颈非常明确}：排序归并每次约 5 秒量级，且需要更高 transient memory。
  \item \keyword{硬件适配结论必须带边界}：Intel 的 \texttt{taskset} 与 Apple 的 QoS profile 不是同一种隔离机制；当前没有 SMT 逻辑核收益证据。
\end{enumerate}
\end{frame}
